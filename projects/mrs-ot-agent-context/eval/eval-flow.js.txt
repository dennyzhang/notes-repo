// deterministic stratified cap: group by type, even-spaced selection within sorted group
function strat(cases, cap) {
  if (cases.length <= cap) return cases
  // BUGFIX: allocate a per-type quota and fill each type independently. The old loop filled types
  // sequentially against a shared cap, so sev+alert consumed the whole cap and posts got 0.
  const byType = {}
  for (const c of cases) (byType[c.type] || (byType[c.type] = [])).push(c)
  const out = []
  for (const t of Object.keys(byType)) {
    const arr = byType[t].slice().sort((a, b) => (a.id < b.id ? -1 : 1))
    const quota = Math.max(1, Math.round((cap * arr.length) / cases.length)) // proportional, guaranteed >=1 per type
    const step = Math.max(1, Math.floor(arr.length / quota))
    for (let i = 0, taken = 0; i < arr.length && taken < quota; i += step, taken++) out.push(arr[i])
  }
  return out.slice(0, cap)
}

// ---- Tier-1 FITNESS FUNCTION (what evolution optimizes) ----
// Principle: no human in the loop -> reward CALIBRATED competence; confident-wrong is the worst outcome.
//   calibration .35  detection_recall .20  scoping_precision .15  owner .15  decisiveness .15
// Hard gate: any hallucination caps that case's contribution to 0.
// detection_recall / scoping_precision need a negative+noise corpus (not built yet) -> null: EXCLUDED and
//   flagged in missing_terms (composite renormalized over present weight); never silently faked.
// Generalization reported SEPARATELY (leak-adjusted proxy until temporal hold-out) — never folded in.
const W = { calibration: 0.35, detection_recall: 0.20, scoping_precision: 0.15, owner: 0.15, decisiveness: 0.15 }
const sumW = (keys) => keys.reduce((s, k) => s + W[k], 0)
const isActionable = (v) => !!v && !/needs[_ ]?investigation|unknown|unclassified|insufficient|undetermined/i.test(v)

function fitnessOver(subset, opts = {}) {
  const n = subset.length
  if (!n) return { composite: 0, dims: {}, missing: Object.keys(W), n: 0, gated: 0 }
  let calib = 0, owner = 0, dec = 0, gated = 0
  // detection_recall / scoping_precision (2026-06-14): computed from the EXISTING gold-set verdict
  // labels — the "negative+noise corpus not built yet" note was stale. The gold set already carries
  // both real-failure cases (the agent can MISS) and noise/no-action cases (the agent can FALSE-POSITIVE).
  //   detection_recall  = verdict-correctness on REAL-failure ground-truth cases (catch the real ones)
  //   scoping_precision = verdict-correctness on NOISE/no-action ground-truth cases (drop the fake ones)
  // Hallucination-gated cases count as a MISS (0) in their class denominator — a hallucinated call is
  // not a correct detection/scoping. Unknown/unclassifiable verdicts are excluded from both.
  const NOISE_RE = /no.?action|noise|transient|detector.?broken|threshold.?misfit|out.?of.?scope|duplicate|wont.?fix|benign/i
  const REAL_RE = /real|recurring|failure|confirmed/i
  let realN = 0, realHit = 0, noiseN = 0, noiseHit = 0
  for (const r of subset) {
    const gv = (r.ground_truth && r.ground_truth.verdict) || ''
    const isNoise = NOISE_RE.test(gv)
    const isReal = !isNoise && REAL_RE.test(gv)
    const gateMiss = r.grade.hallucination >= 1
    if (isReal) { realN++; if (!gateMiss) realHit += (r.grade.verdict_calibrated || 0) }
    else if (isNoise) { noiseN++; if (!gateMiss) noiseHit += (r.grade.verdict_calibrated || 0) }
    if (gateMiss) { gated++; continue } // hard gate: 0 contribution to every per-case term
    const rc = r.grade.root_cause_match
    const conf = Math.max(0, Math.min(1, Number(r.diagnosis.confidence) || 0))
    calib += 1 - Math.pow(rc - conf, 2) // proper scoring: right+confident best; wrong+confident worst; honest-uncertain unpunished
    owner += r.grade.owner_match
    // decisiveness: prefer next_action_quality (v2) when present; fall back to isActionable verdict check (v1)
    dec += (r.grade.next_action_quality != null) ? r.grade.next_action_quality : (isActionable(r.diagnosis.verdict) ? 1 : 0)
  }
  const dims = {
    calibration: +(calib / n).toFixed(3), // gated cases sit in denominator -> hallucination drags every dim down
    owner: +(owner / n).toFixed(3),
    decisiveness: +(dec / n).toFixed(3),
    detection_recall: realN ? +(realHit / realN).toFixed(3) : (opts.detection_recall ?? null),
    scoping_precision: noiseN ? +(noiseHit / noiseN).toFixed(3) : (opts.scoping_precision ?? null),
  }
  let num = 0, den = 0; const missing = []
  for (const k of Object.keys(W)) {
    if (dims[k] == null) { missing.push(k); continue }
    num += W[k] * dims[k]; den += W[k]
  }
  return { composite: den ? +(num / den).toFixed(3) : 0, dims, missing, n, gated, recall_n: realN, precision_n: noiseN }
}

const coverage = fitnessOver(rows) // all cases (memorization allowed)
const generalization = fitnessOver(rows.filter((r) => !r.grade.leak_suspect)) // leak-adjusted proxy
const fitness = {
  ...diagnostics,
  composite: coverage.composite, // headline; renormalized over present weight
  dimensions: coverage.dims,
  weight_covered: +(sumW(Object.keys(W)) - sumW(coverage.missing)).toFixed(2),
  missing_terms: coverage.missing, // flagged, not dropped (needs negative+noise corpus)
  cases_gated_by_hallucination: coverage.gated,
  generalization_composite: generalization.composite,
  generalization_gap: +(coverage.composite - generalization.composite).toFixed(3), // coverage - held-out proxy
}

const byType = {}
for (const r of rows) {
  const t = r.type
  byType[t] = byType[t] || { n: 0, rc: 0 }
  byType[t].n++
  byType[t].rc += r.grade.root_cause_match
}
for (const t of Object.keys(byType)) byType[t].root_cause_accuracy = +(byType[t].rc / byType[t].n).toFixed(3)

// per-component fitness — divide-and-conquer: which component drags the score / hallucinates most?
const byComponent = {}
for (const r of rows) {
  const k = r.grade.component || 'other'
  byComponent[k] = byComponent[k] || { n: 0, rc: 0, halluc: 0 }
  byComponent[k].n++
  byComponent[k].rc += r.grade.root_cause_match
  byComponent[k].halluc += r.grade.hallucination
}
for (const k of Object.keys(byComponent)) {
  byComponent[k].root_cause_accuracy = +(byComponent[k].rc / byComponent[k].n).toFixed(3)
  byComponent[k].hallucination_rate = +(byComponent[k].halluc / byComponent[k].n).toFixed(3)
}
const weakest = Object.entries(byComponent)
  .filter(([, v]) => v.n >= 2)
  .sort((a, b) => a[1].root_cause_accuracy - b[1].root_cause_accuracy)[0]
const weakest_component = weakest ? { component: weakest[0], ...weakest[1] } : null

// failures = evolve targets: low root-cause match, sorted worst-first
const failures = rows
  .filter((r) => r.grade.root_cause_match < 1 || r.grade.hallucination >= 1)
  .sort((a, b) => a.grade.root_cause_match - b.grade.root_cause_match)
  .map((r) => ({
    id: r.id, type: r.type,
    rc_match: r.grade.root_cause_match,
    hallucination: r.grade.hallucination,
    leak_suspect: r.grade.leak_suspect,
    ground_truth_root_cause: r.ground_truth.root_cause,
    agent_root_cause: r.diagnosis.root_cause,
    why: r.grade.rationale,
  }))

log(`Fitness = ${fitness.composite} over ${(fitness.weight_covered * 100).toFixed(0)}% weight (missing: ${fitness.missing_terms.join(', ') || 'none'}); generalization ${fitness.generalization_composite} (gap ${fitness.generalization_gap}); ${fitness.cases_gated_by_hallucination} cases hallucination-gated; n=${rows.length}`)

// machine-checkable outcome marker — the verifiable record gap #1 was missing. The caller appends
// this (plus the fitness dims) to eval-flow-history.jsonl via `eval-stats.sh append`. A graded daemon
// run is now PROVABLE from the history file: source=="daemon" && composite!=null && n>0.
const run_marker = {
  source: RUN_SOURCE,          // "daemon" | "interactive" — caller-threaded; default interactive
  wf: RUN_WF,                  // Workflow run id if the caller passed args.wf, else null (never faked)
  timestamp: RUN_TS,          // ISO-8601, run start
  composite: fitness.composite,
  n: fitness.n_evaluated,
  gold_version: goldVersion,
  inject_rule: (typeof args !== 'undefined' && args && args.inject_rule) || null,
  // sharding provenance (gap #1): a shard is a PARTIAL batch — eval-stats.sh MUST exclude partial
  // rows from the baseline/noise-band so a small/degenerate shard can never masquerade as a full
  // composite (the 0.209/n=5 data-integrity hole). A whole-corpus run is partial:false.
  partial: shardInfo != null,
  shard: shardInfo ? shardInfo.shard : null,
  shard_size: shardInfo ? shardInfo.shard_size : null,
  num_shards: shardInfo ? shardInfo.num_shards : null,
}
log(`run_marker: source=${run_marker.source} partial=${run_marker.partial}${shardInfo ? ' shard=' + run_marker.shard + '/' + (run_marker.num_shards - 1) : ''} composite=${run_marker.composite} n=${run_marker.n} wf=${run_marker.wf || '-'} ts=${run_marker.timestamp}`)

return {
  run_marker,
  // top-level mirrors so `eval-stats.sh append` (and any jsonl reader) finds them without nesting:
  source: RUN_SOURCE,
  wf: RUN_WF,
  timestamp: RUN_TS,
  partial: shardInfo != null,
  shard: shardInfo ? shardInfo.shard : null,
  shard_size: shardInfo ? shardInfo.shard_size : null,
  num_shards: shardInfo ? shardInfo.num_shards : null,
  gold_version: goldVersion,
  corpus: { total_files: files.length, curated: allCases.length, eligible: eligible.length, evaluated: rows.length, harness_excluded: harnessExcluded.length },
  fitness,
  by_type: byType,
  by_component: byComponent,
  weakest_component,
  failures,
  harness_excluded: harnessExcluded.map((r) => r.id),
  all_rows: rows,
}
